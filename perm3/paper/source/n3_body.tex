\subsection{定义与结论}

令
\[
  V=\Span_{\C}\{x_{ij}:0\leq i,j\leq2\},\qquad \dim V=9,
\]
并定义
\[
  \perm_3=
  \sum_{\sigma\in S_3}
  x_{0,\sigma(0)}x_{1,\sigma(1)}x_{2,\sigma(2)}
  \in\Sym^3V.
\]

\begin{definition}
三次型 \(f\in\Sym^3V\) 的 Chow 秩是最小整数 \(r\)，使得
\[
  f=\sum_{s=1}^{r}\ell_{s1}\ell_{s2}\ell_{s3},
  \qquad \ell_{st}\in V.
\]
\end{definition}

\begin{theorem}[主定理]\label{n3:thm:main}
\[
  \boxed{\ChowRank(\perm_3)=4}.
\]
\end{theorem}

\subsection{催化映射与 Koszul 展平}

对任意三次型 \(f\in\Sym^3V\)，定义催化映射
\[
  C_{1,2}(f):V^*\longrightarrow\Sym^2V,
  \qquad D\longmapsto D(f),
\]
并记其像为
\[
  E_f:=\im C_{1,2}(f).
\]
定义 Koszul 微分
\[
  \delta:\Sym^2V\otimes V\longrightarrow
  V\otimes\bigwedge\nolimits^2V
\]
为
\begin{equation}\label{n3:eq:delta}
  \delta((uv)\otimes w)
  =v\otimes(u\wedge w)+u\otimes(v\wedge w),
\end{equation}
并线性延拓。相应的 Koszul 展平为
\[
  K(f):=
  \delta\circ\bigl(C_{1,2}(f)\otimes\id_V\bigr).
\]
于是
\begin{equation}\label{n3:eq:image}
  \im K(f)=\delta(E_f\otimes V).
\end{equation}

\subsubsection{Koszul 核与一阶延拓}

对任意子空间 \(E\subseteq\Sym^2V\)，定义它的一阶延拓
\[
  E^{(1)}:=
  \left\{g\in\Sym^3V:
  \frac{\partial g}{\partial x_{ij}}\in E
  \text{ 对所有 }i,j\right\}.
\]

\begin{lemma}\label{n3:lem:kernel-prolongation}
有自然同构
\[
  \ker\bigl(\delta|_{E\otimes V}\bigr)\simeq E^{(1)}.
\]
因此
\[
  \rank\delta(E\otimes V)
  =9\dim E-\dim E^{(1)}.
\]
\end{lemma}

\begin{proof}
定义极化映射
\[
  \iota:\Sym^3V\longrightarrow\Sym^2V\otimes V,
  \qquad
  \iota(g)=\frac13\sum_{i,j}
  \frac{\partial g}{\partial x_{ij}}\otimes x_{ij}.
\]
Euler 恒等式说明，张量乘法映射把 \(\iota(g)\) 送回 \(g\)，故
\(\iota\) 是单射。将式~\eqref{n3:eq:delta} 用于三次单项式，可见所有
楔积项成对相消，所以
\[
  \delta\circ\iota=0.
\]

反过来，把 \(z\in\Sym^2V\otimes V\) 的系数写成 \(z_{abc}\)。由于
前两个张量指标来自 \(\Sym^2V\)，已有 \(z_{abc}=z_{bac}\)；条件
\(\delta(z)=0\) 又强迫 \(z_{abc}=z_{acb}\)。这两个换位生成全部
三个指标的置换，故 \(z_{abc}\) 完全对称，从而 \(z=\iota(g)\) 对某个
唯一的 \(g\in\Sym^3V\) 成立。因此
\[
  \ker\delta=\iota(\Sym^3V).
\]
再与 \(E\otimes V\) 相交，恰好要求 \(g\) 的所有一阶偏导属于 \(E\)，
即得到 \(E^{(1)}\)。最后使用秩--零化度公式和 \(\dim V=9\) 即得秩
公式。
\end{proof}

\subsection{永久多项式的展平秩}

记 \(E:=E_{\perm_3}\)。对 \(x_{ij}\) 求导，会得到删去第 \(i\) 行和
第 \(j\) 列后的 \(2\times2\) 子永久式。因此
\begin{equation}\label{n3:eq:E}
  E=
  \Span\left\{
  x_{ia}x_{jb}+x_{ib}x_{ja}:
  0\leq i<j\leq2,\ 0\leq a<b\leq2
  \right\}.
\end{equation}
不同的行对、列对具有互不混淆的单项式支撑，故这九个二次型线性
无关：
\begin{equation}\label{n3:eq:E-dim}
  \dim E=\binom32^2=9.
\end{equation}

\begin{proposition}\label{n3:prop:prolongation}
\[
  E^{(1)}=\Span(\perm_3).
\]
特别地，\(\dim E^{(1)}=1\)。
\end{proposition}

\begin{proof}
首先，\(\perm_3\) 的每个一阶偏导都属于 \(E\)，所以
\(\perm_3\in E^{(1)}\)。

反之，取 \(g\in E^{(1)}\)。由于式~\eqref{n3:eq:E} 中的每个二次单项式
都使用互异的变量行和互异的变量列，条件
\(\partial g/\partial x_{ij}\in E\) 强迫 \(g\) 的每个非零三次
单项式也使用三个互异行和三个互异列。因此
\[
  g=\sum_{\sigma\in S_3}c_{\sigma}
  x_{0,\sigma(0)}x_{1,\sigma(1)}x_{2,\sigma(2)}.
\]

固定其中一个变量并对它求导，剩余两行、两列上只出现两个可能匹配。
而 \(E\) 在该行对、列对上的部分是一维的，由这两个匹配之和生成；所以
相应的两个系数必须相等。换言之，若两个置换相差一个换位，则它们的
系数相等。换位生成整个 \(S_3\)，故所有 \(c_{\sigma}\) 相等。因此
\(g\) 是 \(\perm_3\) 的标量倍数。
\end{proof}

\begin{corollary}\label{n3:cor:perm-rank}
\[
  \rank K(\perm_3)=80.
\]
\end{corollary}

\begin{proof}
由引理~\ref{n3:lem:kernel-prolongation}、式~\eqref{n3:eq:E-dim} 和命题
~\ref{n3:prop:prolongation}，
\[
  \rank K(\perm_3)
  =9\cdot9-1=80.
\]
\end{proof}

\subsection{单个 Chow 项的统一秩上界}

\begin{proposition}\label{n3:prop:term-rank}
对任意三次 Chow 项 \(T=\ell_1\ell_2\ell_3\)，有
\[
  \rank K(T)\leq26.
\]
当三个因子线性无关时，等号成立。
\end{proposition}

\begin{proof}
记 \(F:=E_T\)。对 \(T\) 求一次导数可知
\[
  F\subseteq
  \Span\{\ell_1\ell_2,\ell_1\ell_3,\ell_2\ell_3\},
  \qquad \dim F\leq3.
\]
若 \(T\neq0\)，则 \(T\in F^{(1)}\)，所以
\(\dim F^{(1)}\geq1\)。引理~\ref{n3:lem:kernel-prolongation} 给出
\[
  \rank K(T)
  =9\dim F-\dim F^{(1)}
  \leq9\cdot3-1=26.
\]
零 Chow 项的结论显然成立。

若三个因子线性无关，经 \(GL(V)\) 变换可令
\(T=x_1x_2x_3\)。此时
\[
  F=\Span\{x_1x_2,x_1x_3,x_2x_3\},
  \qquad F^{(1)}=\Span(x_1x_2x_3),
\]
故秩恰为 \(27-1=26\)。
\end{proof}

\subsection{下界与四项分解}

\begin{theorem}\label{n3:thm:lower}
\[
  \ChowRank(\perm_3)\geq4.
\]
\end{theorem}

\begin{proof}
若
\[
  \perm_3=T_1+\cdots+T_r
\]
是 \(r\) 个 Chow 项之和，则 \(K\) 关于多项式线性，矩阵秩又满足
次可加性。因此由推论~\ref{n3:cor:perm-rank} 和命题
~\ref{n3:prop:term-rank}，
\[
  80=\rank K(\perm_3)
  \leq\sum_{s=1}^{r}\rank K(T_s)
  \leq26r.
\]
所以
\[
  r\geq\left\lceil\frac{80}{26}\right\rceil=4.
\]
\end{proof}

另一方面，三阶 Glynn 恒等式给出显式分解
\begin{align}\label{n3:eq:glynn}
4\perm_3={}&
\prod_{j=0}^{2}(x_{0j}+x_{1j}+x_{2j})\notag\\
&-\prod_{j=0}^{2}(x_{0j}+x_{1j}-x_{2j})\notag\\
&-\prod_{j=0}^{2}(x_{0j}-x_{1j}+x_{2j})\\
&+\prod_{j=0}^{2}(x_{0j}-x_{1j}-x_{2j}).\notag
\end{align}
右侧每一行都是三个线性形式的乘积，标量和符号可以吸收到其中一个
线性因子中。因此
\[
  \ChowRank(\perm_3)\leq4.
\]
与定理~\ref{n3:thm:lower} 合并，证明主定理~\ref{n3:thm:main}。
